/* BC Radio Control Room: on the air, songs, add songs, featured times, history. No dependencies. */
(function () {
  'use strict';
  var $ = function (s, r) { return (r || document).querySelector(s); };
  var $$ = function (s, r) { return [].slice.call((r || document).querySelectorAll(s)); };
  var S = { me: null, songs: [], byId: {}, features: [], sel: {}, filter: { q: '', show: '', section: '', sort: 'az' }, edit: null, booth: null, skew: 0, tab: 'booth' };

  // ---------- basics ----------
  function api(path, opts) {
    opts = opts || {};
    var h = { 'x-cr': '1' };
    var body = opts.body;
    if (body && !(body instanceof Blob) && !(body instanceof ArrayBuffer)) { h['content-type'] = 'application/json'; body = JSON.stringify(body); }
    if (opts.type) h['content-type'] = opts.type;
    return fetch('/api' + path, { method: opts.method || 'GET', headers: h, body: body, credentials: 'same-origin' }).then(function (r) {
      if (r.status === 403 && r.headers.get('x-cr-auth')) { location.reload(); throw new Error('Signed out'); }
      return r.json().catch(function () { return {}; }).then(function (j) { if (!r.ok) throw new Error(j.error || ('Something went wrong (' + r.status + ')')); return j; });
    });
  }
  function el(tag, attrs, kids) {
    var n = document.createElement(tag);
    if (attrs) for (var k in attrs) {
      var v = attrs[k];
      if (v == null || v === false) continue;
      if (k === 'text') n.textContent = v; else if (k === 'html') n.innerHTML = v;
      else if (k === 'style') n.style.cssText = v;   // CSSOM: allowed by the page's strict CSP (style attributes are not)
      else if (k.slice(0, 2) === 'on') n.addEventListener(k.slice(2), v);
      else n.setAttribute(k, v === true ? '' : v);
    }
    (kids || []).forEach(function (c) { if (c != null) n.appendChild(typeof c === 'string' ? document.createTextNode(c) : c); });
    return n;
  }
  var toastT;
  function toast(msg, undoFn) {
    var t = $('#toast');
    t.textContent = msg;
    if (undoFn) t.appendChild(el('button', { type: 'button', text: 'Undo', onclick: function () { t.hidden = true; undoFn(); } }));
    t.hidden = false;
    clearTimeout(toastT); toastT = setTimeout(function () { t.hidden = true; }, undoFn ? 9000 : 4500);
  }
  // Big plain-words confirm (used before anything listeners hear right away).
  function confirmBox(title, text, yes) {
    return new Promise(function (resolve) {
      $('#cf-title').textContent = title; $('#cf-body').textContent = text || ''; $('#cf-yes').textContent = yes || 'Yes';
      $('#cf-scrim').hidden = false; setTimeout(function () { $('#cf-yes').focus(); }, 30);
      function done(v) { $('#cf-scrim').hidden = true; $('#cf-yes').onclick = $('#cf-no').onclick = null; document.removeEventListener('keydown', key); resolve(v); }
      function key(e) { if (e.key === 'Escape') done(false); }
      $('#cf-yes').onclick = function () { done(true); };
      $('#cf-no').onclick = function () { done(false); };
      document.addEventListener('keydown', key);
    });
  }
  function nowMs() { return Date.now() + S.skew; }
  function mmss(ms) { ms = Math.max(0, Math.round(ms / 1000)); return Math.floor(ms / 60) + ':' + String(ms % 60).padStart(2, '0'); }
  var TZ = 'America/Los_Angeles';
  function clock(ms) { return new Date(ms).toLocaleTimeString('en-US', { timeZone: TZ, hour: 'numeric', minute: '2-digit' }); }
  function when(ms) {
    if (!ms) return 'never';
    var d = nowMs() - ms;
    if (d < 0) return 'coming up';
    if (d < 3600e3) return Math.max(1, Math.round(d / 60000)) + ' min ago';
    if (d < 86400e3) return Math.round(d / 3600e3) + ' h ago';
    if (d < 30 * 86400e3) return Math.round(d / 86400e3) + ' days ago';
    return new Date(ms).toLocaleDateString('en-US', { timeZone: TZ, month: 'short', day: 'numeric', year: 'numeric' });
  }
  function fullWhen(ms) { return new Date(ms).toLocaleString('en-US', { timeZone: TZ, month: 'short', day: 'numeric', hour: 'numeric', minute: '2-digit' }); }
  function natural(a, b) { return String(a).localeCompare(String(b), 'en', { sensitivity: 'base', numeric: true }); }
  function plural(n, w) { return n + ' ' + w + (n === 1 ? '' : 's'); }

  // ---------- tabs ----------
  function show(tab) {
    S.tab = tab;
    $$('.tabb').forEach(function (b) { b.classList.toggle('on', b.getAttribute('data-tab') === tab); });
    $$('[data-panel]').forEach(function (p) { p.hidden = p.getAttribute('data-panel') !== tab; });
    if (tab === 'booth') loadBooth();
    if (tab === 'activity') loadActivity();
    if (tab === 'featured') loadFeatures();
    if (location.hash !== '#' + tab) history.replaceState(null, '', '#' + tab);
  }
  $$('.tabb').forEach(function (b) { b.addEventListener('click', function () { show(b.getAttribute('data-tab')); }); });
  var TABS = ['booth', 'songs', 'upload', 'featured', 'activity'];
  window.addEventListener('hashchange', function () { var t = location.hash.slice(1); if (S.me && TABS.indexOf(t) > -1 && t !== S.tab) show(t); });

  // ---------- songs ----------
  function index() { S.byId = {}; S.songs.forEach(function (s) { S.byId[s.id] = s; }); }
  function loadSongs() { return api('/songs').then(function (j) { S.skew = j.now - Date.now(); S.songs = j.songs; index(); renderList(); fillLists(); }); }
  function featuredIds() {
    var m = {};
    S.features.forEach(function (f) { if (f.active && !f.finished) f.song_ids.forEach(function (id) { (m[id] = m[id] || []).push(f.name); }); });
    return m;
  }
  function filtered() {
    var f = S.filter, words = f.q.toLowerCase().split(/\s+/).filter(Boolean), fids = featuredIds();
    var rows = S.songs.filter(function (s) {
      if (f.show === 'on' && !s.enabled) return false;
      if (f.show === 'off' && s.enabled) return false;
      if (f.show === 'never' && s.plays) return false;
      if (f.show === 'featured' && !fids[s.id]) return false;
      if (f.show === 'new' && !s.uploaded) return false;
      if (f.show === 'noreq' && s.requestable) return false;
      if (f.section && s.section !== f.section) return false;
      if (!words.length) return true;
      var hay = [s.title, s.album, s.artist, s.year, s.section, s.raw_title].join(' ').toLowerCase();
      return words.every(function (w) { return hay.indexOf(w) > -1; });
    });
    var by = {
      az: function (a, b) { return natural(a.title, b.title); },
      album: function (a, b) { return natural(a.album, b.album) || ((a.no == null ? 1e9 : a.no) - (b.no == null ? 1e9 : b.no)) || natural(a.title, b.title); },
      plays: function (a, b) { return b.plays - a.plays || natural(a.title, b.title); },
      fewest: function (a, b) { return a.plays - b.plays || natural(a.title, b.title); },
      recent: function (a, b) { return b.last_played_ms - a.last_played_ms; },
      longago: function (a, b) { return a.last_played_ms - b.last_played_ms; },
      added: function (a, b) { return b.added - a.added; },
      long: function (a, b) { return b.dur_ms - a.dur_ms; },
    };
    rows.sort(by[f.sort] || by.az);
    return rows;
  }
  function renderStats() {
    var L = S.songs, on = L.filter(function (s) { return s.enabled; }).length, fids = featuredIds();
    var never = L.filter(function (s) { return !s.plays; }).length, feat = L.filter(function (s) { return fids[s.id]; }).length;
    var hours = L.filter(function (s) { return s.enabled; }).reduce(function (a, s) { return a + s.dur_ms; }, 0) / 3600e3;
    var cards = [['on', on, 'On the air (' + hours.toFixed(1) + ' h)'], ['off', L.length - on, 'Switched off'], ['never', never, 'Never played'], ['featured', feat, 'In a featured time'], ['new', L.filter(function (s) { return s.uploaded; }).length, 'Added here']];
    var box = $('#stats'); box.textContent = '';
    cards.forEach(function (c) {
      box.appendChild(el('button', { type: 'button', class: 'stat' + (S.filter.show === c[0] ? ' on' : ''),
        onclick: function () { S.filter.show = S.filter.show === c[0] ? '' : c[0]; $('#f-show').value = S.filter.show; renderList(); } },
        [el('div', { class: 'stat-n', text: String(c[1]) }), el('div', { class: 'stat-l', text: c[2] })]));
    });
    $('#t-count').textContent = String(L.length);
  }
  var renderCap = 120;
  function renderList() {
    renderStats();
    var rows = filtered(), box = $('#rows'), fids = featuredIds();
    box.textContent = '';
    if (!rows.length) box.appendChild(el('div', { class: 'empty', text: 'No songs match.' }));
    rows.slice(0, renderCap).forEach(function (s) {
      var ck = el('input', { type: 'checkbox', 'aria-label': 'Select ' + s.title });
      ck.checked = !!S.sel[s.id];
      ck.addEventListener('click', function (e) { e.stopPropagation(); if (ck.checked) S.sel[s.id] = 1; else delete S.sel[s.id]; renderBulk(); });
      var onBtn = el('button', { type: 'button', class: 'tog', title: s.enabled ? 'Switch off (take out of the rotation)' : 'Switch on (put in the rotation)',
        onclick: function (e) { e.stopPropagation(); quickToggle(s); } },
        [el('span', { class: 'pill ' + (s.enabled ? 'on' : 'off'), text: s.enabled ? 'On' : 'Off' })]);
      var flags = el('div', { class: 'flags' }, [onBtn,
        s.uploaded && !s.plays && !s.enabled ? el('span', { class: 'pill new', text: 'New' }) : null,
        fids[s.id] ? el('span', { class: 'pill feat', text: 'Featured', title: fids[s.id].join(', ') }) : null,
        !s.requestable ? el('span', { class: 'pill warn', text: 'No requests' }) : null]);
      var row = el('div', { class: 'lrow' + (s.enabled ? '' : ' off'), role: 'row', tabindex: '0', 'data-id': s.id, onclick: function () { openEdit(s.id); },
        onkeydown: function (e) { if (e.key === 'Enter') openEdit(s.id); } }, [
        el('label', { class: 'ck', onclick: function (e) { e.stopPropagation(); } }, [ck]),
        el('div', { class: 'l-main' }, [
          el('img', { class: 'l-thumb', src: s.art, alt: '', loading: 'lazy' }),
          el('div', { class: 'l-txt' }, [el('div', { class: 'l-name', text: s.title }), el('div', { class: 'l-sub', text: [s.album, s.artist !== 'Bernie Chiaravalle' ? s.artist : null, s.year].filter(Boolean).join(' · ') || s.section || '' })]),
        ]),
        el('div', { class: 'c-len', text: mmss(s.dur_ms) }),
        el('div', { class: 'c-plays', text: String(s.plays) }),
        el('div', { class: 'c-last', text: when(s.last_played_ms) }),
        flags,
      ]);
      box.appendChild(row);
    });
    if (rows.length > renderCap) box.appendChild(el('div', { class: 'empty' }, [el('button', { class: 'btn', type: 'button', text: 'Show all ' + rows.length, onclick: function () { renderCap = 1e9; renderList(); } })]));
    $('#list-foot').textContent = 'Showing ' + Math.min(rows.length, renderCap) + ' of ' + rows.length + (rows.length !== S.songs.length ? ' (' + S.songs.length + ' songs in total)' : '') + '. Tap a song to edit it, change its cover, or play it.';
    $('#ck-all').checked = rows.length > 0 && rows.every(function (s) { return S.sel[s.id]; });
    renderBulk();
  }
  function renderBulk() {
    var n = Object.keys(S.sel).length;
    $('#bulk').hidden = !n;
    $('#bulk-n').textContent = plural(n, 'song') + ' selected';
  }
  function replace(s) { var i = S.songs.findIndex(function (x) { return x.id === s.id; }); if (i < 0) S.songs.push(s); else S.songs[i] = s; S.byId[s.id] = s; }
  function quickToggle(s) {
    var to = !s.enabled;
    var go = to ? Promise.resolve(true) : confirmBox('Switch off "' + s.title + '"?', 'It stops being picked for the radio. If it is coming up soon, the songs coming up are planned again (the song playing now is not interrupted). You can undo this.', 'Switch it off');
    go.then(function (ok) {
      if (!ok) return;
      var call = to && s.uploaded && !s.plays ? api('/songs/' + s.id + '/publish', { method: 'POST' }) : api('/songs/' + s.id, { method: 'PATCH', body: { enabled: to } });
      call.then(function (j) {
        replace(j.song); renderList();
        toast((to ? 'Switched on "' : 'Switched off "') + s.title + '"' + (j.replanned ? '. The songs coming up were planned again.' : ''), function () {
          api('/songs/' + s.id, { method: 'PATCH', body: { enabled: !to } }).then(function (k) { replace(k.song); renderList(); toast('Undone'); });
        });
      }).catch(function (e) { toast(e.message); });
    });
  }
  $('#q').addEventListener('input', function () { S.filter.q = this.value; renderCap = 120; renderList(); });
  $('#f-show').addEventListener('change', function () { S.filter.show = this.value; renderList(); });
  $('#f-section').addEventListener('change', function () { S.filter.section = this.value; renderList(); });
  $('#f-sort').addEventListener('change', function () { S.filter.sort = this.value; renderList(); });
  $('#ck-all').addEventListener('change', function () {
    var on = this.checked;
    filtered().forEach(function (s) { if (on) S.sel[s.id] = 1; else delete S.sel[s.id]; });
    renderList();
  });
  $('#bulk-clear').addEventListener('click', function () { S.sel = {}; renderList(); });
  $$('[data-bulk]').forEach(function (b) {
    b.addEventListener('click', function () {
      var action = b.getAttribute('data-bulk'), ids = Object.keys(S.sel);
      var words = { enable: 'Switch on', disable: 'Switch off', requests_on: 'Allow requests for', requests_off: 'Stop requests for' }[action];
      var go = action === 'disable' ? confirmBox(words + ' ' + plural(ids.length, 'song') + '?', 'They stop being picked for the radio. Songs coming up are planned again if needed. You can undo this from History.', words) : Promise.resolve(true);
      go.then(function (ok) {
        if (!ok) return;
        api('/songs/bulk', { method: 'POST', body: { action: action, ids: ids } }).then(function (j) {
          S.sel = {}; toast(plural(j.changed, 'song') + ' updated' + (j.replanned ? '. The songs coming up were planned again.' : '')); return loadSongs();
        }).catch(function (e) { toast(e.message); });
      });
    });
  });
  function fillLists() {
    var albums = {}, artists = {};
    S.songs.forEach(function (s) { if (s.album) albums[s.album] = 1; if (s.artist) artists[s.artist] = 1; });
    var a = $('#albums'), b = $('#artists'); a.textContent = ''; b.textContent = '';
    Object.keys(albums).sort(natural).forEach(function (x) { a.appendChild(el('option', { value: x })); });
    Object.keys(artists).sort(natural).forEach(function (x) { b.appendChild(el('option', { value: x })); });
  }

  // ---------- song drawer ----------
  var form = $('#dr-form');
  function openEdit(id) {
    var s = S.byId[id]; if (!s) return;
    S.edit = s;
    $('#dr-title').textContent = s.title;
    $('#dr-sub').textContent = [s.album, s.artist].filter(Boolean).join(' · ');
    $('#dr-art').src = s.art;
    var au = $('#dr-audio'); au.pause(); au.src = s.url;
    ['title', 'album', 'artist', 'year', 'no'].forEach(function (k) { form.elements[k].value = s[k] == null ? '' : s[k]; });
    var sec = $('#dr-section'); sec.textContent = ''; sec.appendChild(el('option', { value: '', text: '(none)' }));
    S.me.sections.forEach(function (x) { sec.appendChild(el('option', { value: x, text: x })); });
    sec.value = s.section || '';
    form.elements.enabled.checked = s.enabled;
    form.elements.requestable.checked = s.requestable;
    $('#dr-new').hidden = !(s.uploaded && !s.enabled && !s.plays);
    $('#dr-delete').hidden = !(s.uploaded && !s.enabled && !s.plays);
    var facts = $('#dr-facts'); facts.textContent = '';
    [['Length', mmss(s.dur_ms)], ['Times played', String(s.plays)], ['Last played', s.last_played_ms ? (s.last_played_ms > nowMs() ? 'coming up at ' + clock(s.last_played_ms) : fullWhen(s.last_played_ms)) : 'never'], ['Song id', s.id]].forEach(function (f) {
      facts.appendChild(el('div', null, [f[0] + ': ', el('b', { text: f[1] })]));
    });
    $('#dr-msg').textContent = ''; $('#dr-msg').className = 'msg';
    $('#scrim').hidden = false; $('#drawer').classList.add('open'); $('#drawer').setAttribute('aria-hidden', 'false');
  }
  function closeEdit() {
    $('#dr-audio').pause();
    $('#scrim').hidden = true; $('#drawer').classList.remove('open'); $('#drawer').setAttribute('aria-hidden', 'true');
    $('#fdrawer').classList.remove('open'); $('#fdrawer').setAttribute('aria-hidden', 'true');
    S.edit = null; S.fedit = null;
  }
  $('#dr-close').addEventListener('click', closeEdit);
  $('#dr-cancel').addEventListener('click', closeEdit);
  $('#scrim').addEventListener('click', closeEdit);
  document.addEventListener('keydown', function (e) { if (e.key === 'Escape' && (S.edit || S.fedit) && $('#cf-scrim').hidden) closeEdit(); });
  function drMsg(t, cls) { $('#dr-msg').textContent = t; $('#dr-msg').className = 'msg' + (cls ? ' ' + cls : ''); }

  $('#dr-save').addEventListener('click', function () {
    var s = S.edit; if (!s) return;
    if (!form.elements.title.value.trim()) { drMsg('A title is required', 'bad'); form.elements.title.focus(); return; }
    var b = { title: form.elements.title.value, album: form.elements.album.value, artist: form.elements.artist.value, year: form.elements.year.value, no: form.elements.no.value, section: form.elements.section.value, requestable: form.elements.requestable.checked };
    var turningOff = s.enabled && !form.elements.enabled.checked, turningOn = !s.enabled && form.elements.enabled.checked;
    var go = turningOff ? confirmBox('Switch off "' + s.title + '"?', 'It stops being picked for the radio. You can undo this.', 'Switch it off') : Promise.resolve(true);
    go.then(function (ok) {
      if (!ok) return;
      if (!turningOn) b.enabled = form.elements.enabled.checked;
      $('#dr-save').disabled = true;
      api('/songs/' + s.id, { method: 'PATCH', body: b }).then(function (j) {
        replace(j.song);
        return turningOn ? api('/songs/' + s.id + '/publish', { method: 'POST' }).then(function (k) { replace(k.song); return j; }) : j;
      }).then(function (j) {
        renderList(); closeEdit();
        toast(j.changed.length || turningOn ? 'Saved "' + S.byId[s.id].title + '"' + (j.replanned ? '. The songs coming up were planned again.' : '') : 'No changes');
      }).catch(function (e) { drMsg(e.message, 'bad'); }).then(function () { $('#dr-save').disabled = false; });
    });
  });
  $('#dr-publish').addEventListener('click', function () {
    var s = S.edit;
    var b = { title: form.elements.title.value, album: form.elements.album.value, artist: form.elements.artist.value, year: form.elements.year.value, no: form.elements.no.value, section: form.elements.section.value, requestable: form.elements.requestable.checked };
    if (!b.title.trim()) { drMsg('A title is required', 'bad'); return; }
    api('/songs/' + s.id, { method: 'PATCH', body: b }).then(function () { return api('/songs/' + s.id + '/publish', { method: 'POST' }); }).then(function (j) {
      replace(j.song); renderList(); closeEdit(); toast('Published "' + j.song.title + '". It is now in the rotation.');
    }).catch(function (e) { drMsg(e.message, 'bad'); });
  });
  $('#dr-delete').addEventListener('click', function () {
    var s = S.edit;
    confirmBox('Remove "' + s.title + '"?', 'It was never on the air. It comes out of the song list (you can undo this from History).', 'Remove').then(function (ok) {
      if (!ok) return;
      api('/songs/' + s.id, { method: 'DELETE' }).then(function () {
        S.songs = S.songs.filter(function (x) { return x.id !== s.id; }); index(); closeEdit(); renderList(); toast('Removed "' + s.title + '"');
      }).catch(function (e) { drMsg(e.message, 'bad'); });
    });
  });
  $('#dr-cover').addEventListener('change', function () {
    var f = this.files && this.files[0]; this.value = '';
    if (!f) return;
    var s = S.edit;
    drMsg('Preparing the cover…');
    uploadCover(f).then(function (key) {
      return api('/songs/' + s.id, { method: 'PATCH', body: { art: key } });
    }).then(function (j) {
      replace(j.song); $('#dr-art').src = j.song.art; renderList(); drMsg('New cover saved', 'ok');
    }).catch(function (e) { drMsg(e.message, 'bad'); });
  });
  function uploadCover(fileOrBlob) {
    return import('/convert.js').then(function (C) { return C.makeCover(fileOrBlob); }).then(function (c) { return putCover(c); });
  }
  function putCover(c) {
    return api('/art?name=' + c.name + '&v=main', { method: 'POST', body: c.main, type: 'image/jpeg' })
      .then(function () { return api('/art?name=' + c.name + '&v=hd', { method: 'POST', body: c.hd, type: 'image/jpeg' }); })
      .then(function () { return 'art/' + c.name + '.jpg'; });
  }
  function playSong(s, mode) {
    var title = mode === 'now' ? 'Play "' + s.title + '" now?' : 'Play "' + s.title + '" next?';
    var text = mode === 'now'
      ? 'Everyone listening hears it in about 10 seconds. The song playing now stops early.'
      : 'It plays for everyone right after the song that is playing now.';
    return confirmBox(title, text, mode === 'now' ? 'Play it now' : 'Play it next').then(function (ok) {
      if (!ok) return false;
      return api('/booth/play', { method: 'POST', body: { id: s.id, mode: mode } }).then(function (j) {
        toast(mode === 'now' ? '"' + s.title + '" starts for everyone at ' + clock(j.at) : '"' + s.title + '" plays next, at ' + clock(j.at));
        loadBooth();
        return true;
      }).catch(function (e) { toast(e.message); return false; });
    });
  }
  $('#dr-playnow').addEventListener('click', function () { var s = S.edit; playSong(s, 'now').then(function (ok) { if (ok) { closeEdit(); show('booth'); } }); });
  $('#dr-playnext').addEventListener('click', function () { var s = S.edit; playSong(s, 'next').then(function (ok) { if (ok) { closeEdit(); show('booth'); } }); });

  var boothT = null;
  function loadBooth() {
    return api('/booth').then(function (j) { S.skew = j.now - Date.now(); S.booth = j; renderBooth(); }).catch(function (e) { $('#now-title').textContent = e.message; });
  }
  function tags(i) {
    return el('span', { class: 'tags' }, [i.is_dj ? el('span', { class: 'pill live', text: 'Your pick' }) : null, i.is_request ? el('span', { class: 'pill new', text: 'Request' }) : null]);
  }

  }
  function renderBooth() {
    var b = S.booth; if (!b) return;
    var c = b.current;
    if (c) {
      $('#now-art').src = c.track.art_hd || c.track.art;

  function upload(r, onProg) {
    var coverKey = Promise.resolve(null);
    if (r.cover) coverKey = putCover(r.cover);
    return coverKey.then(function (artKey) {
      r.artKey = artKey;
      return api('/uploads', { method: 'POST', body: { size: r.bytes, sha1: r.sha1 } });
    }).then(function (u) {
      var parts = [], n = Math.ceil(r.bytes / u.part_size), i = 0;
      function next() {
        if (i >= n) return Promise.resolve();
        var num = i + 1, blob = r.mp3.slice(i * u.part_size, Math.min(r.bytes, (i + 1) * u.part_size));
        return api('/uploads/' + u.upload + '/parts/' + num, { method: 'PUT', body: blob, type: 'application/octet-stream' }).then(function (p) {
          parts.push(p); i++; onProg(i / n); return next();
        });
      }
      return next().then(function () {
        var m = r.meta;
        return api('/uploads/' + u.upload + '/complete', { method: 'POST', body: { parts: parts, dur_ms: r.dur_ms, meta: {
          title: m.title, album: m.album, artist: m.artist, year: m.year, no: m.no, art: r.artKey, lufs: r.lufs, filename: r.filename || '', section: m.album ? 'Albums' : 'Singles' } } });
      }).catch(function (e) { api('/uploads/' + u.upload + '/abort', { method: 'POST' }).catch(function () {}); throw e; });
    }).then(function (j) { return j.song; });
  }

  function renderPresets(on) {

  function syncMode() {
    var mode = getRadio('mode');
    $('#fd-weights').hidden = mode === 'only';

  function toggleItem(kind, value, on) { var k = kind + '|' + value; if (on) FD.items[k] = { kind: kind, value: value }; else delete FD.items[k]; renderSum(); }
  function renderPick() {
    var box = $('#fd-pick'); box.textContent = '';
    var q = $('#fd-q').value.toLowerCase().trim();
    var match = function (t) { return !q || q.split(/\s+/).every(function (w) { return t.toLowerCase().indexOf(w) > -1; }); };
    function item(kind, value, img, t, s) {
      var ck = el('input', { type: 'checkbox' }); ck.checked = !!FD.items[kind + '|' + value];
      ck.addEventListener('change', function () { toggleItem(kind, value, ck.checked); });
      return el('label', { class: 'pick-item' }, [ck, img ? el('img', { src: img, alt: '', loading: 'lazy' }) : el('span'), el('div', { style: 'min-width:0' }, [el('div', { class: 't', text: t }), el('div', { class: 's', text: s })])]);
    }
    if (FD.ptab === 'albums') {
      albumList().filter(function (a) { return match(a.name); }).forEach(function (a) { box.appendChild(item('album', a.name, a.art, a.name, a.section + ' · ' + plural(a.n, 'song') + (a.off ? ' · ' + a.off + ' switched off' : ''))); });
    } else if (FD.ptab === 'sections') {
      S.me.sections.forEach(function (sec) { var n = S.songs.filter(function (s) { return s.section === sec; }).length; if (match(sec)) box.appendChild(item('section', sec, null, sec, plural(n, 'song'))); });
    } else {
      S.songs.filter(function (s) { return match(s.title + ' ' + s.album); }).sort(function (a, b) { return natural(a.title, b.title); }).slice(0, 200)
        .forEach(function (s) { box.appendChild(item('track', s.id, s.art, s.title, [s.album, mmss(s.dur_ms), s.enabled ? null : 'switched off'].filter(Boolean).join(' · '))); });
    }
    if (!box.children.length) box.appendChild(el('p', { class: 'muted small', style: 'padding:10px 12px;margin:0', text: 'Nothing matches.' }));
    renderSum();
  }
  function resolved() {
    var ids = {};
    Object.keys(FD.items).forEach(function (k) {
      var it = FD.items[k];
      S.songs.forEach(function (s) { if ((it.kind === 'track' && s.id === it.value) || (it.kind === 'album' && s.album === it.value) || (it.kind === 'section' && s.section === it.value)) ids[s.id] = s; });
    });
    return Object.keys(ids).map(function (k) { return ids[k]; });
  }
  function renderSum() {
    var r = resolved(), off = r.filter(function (s) { return !s.enabled; }).length, picks = Object.keys(FD.items).map(function (k) { return itemLabel(FD.items[k]); });
    var mode = getRadio('mode');
    $('#fd-sum').textContent = r.length ? 'Picked: ' + picks.join(', ') + ' → ' + plural(r.length, 'song') + (off ? ' (' + off + ' switched off)' : '') : 'Nothing picked yet.';
    var warn = $('#fd-warn'); warn.hidden = true;
    if (mode === 'more' && off && off === r.length) { warn.hidden = false; warn.textContent = 'All of these songs are switched off, so "play more often" would do nothing. Choose "Let switched-off songs play" instead.'; }
    else if (mode === 'more' && off) { warn.hidden = false; warn.textContent = off + ' of these are switched off and stay off. To let them play during these dates, choose "Let switched-off songs play".'; }
    else if (mode === 'unlock' && !off) { warn.hidden = false; warn.textContent = 'None of these songs are switched off. "Let switched-off songs play" only matters for songs that are off. To play these more often, choose "Play these more often".'; }
    else if (mode === 'only' && r.length && r.length < 15) { warn.hidden = false; warn.textContent = 'Only ' + plural(r.length, 'song') + ': a song never repeats within 6 hours, so when they have all played, the regular mix fills in until they can play again.'; }
  }

        var q = {}; new Intl.DateTimeFormat('en-US', { timeZone: b.tz, year: 'numeric', month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit', hourCycle: 'h23' }).formatToParts(new Date(nowMs())).forEach(function (x) { q[x.type] = x.value; });
        var nowL = q.year + '-' + q.month + '-' + q.day + 'T' + q.hour + ':' + q.minute;
        var inWin = function (s, e) { return s <= soon && e > nowL; };
        if (inWin(b.start_local, b.end_local)) return true;
        if (b.yearly) { var y = +nowL.slice(0, 4); for (var k = -1; k <= 0; k++) { var sy = y + k, ey = sy + (+b.end_local.slice(0, 4) - +b.start_local.slice(0, 4)); if (inWin(sy + b.start_local.slice(4), ey + b.end_local.slice(4))) return true; } }
